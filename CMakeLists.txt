cmake_minimum_required(VERSION 3.22)
project(L2Limiter VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(CTest)
add_executable(dsp_tests Tests/DspTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME DSP COMMAND dsp_tests)
option(L2_BUILD_PLUGIN "Build JUCE plugin and standalone" ON)
if(L2_BUILD_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG d054f0d14dcac387aebda44ce5d792b5e7a625b3 GIT_SHALLOW FALSE)
 FetchContent_MakeAvailable(JUCE)
 set(formats VST3 Standalone)
 if(APPLE)
  list(APPEND formats AU)
 endif()
 juce_add_plugin(L2Limiter COMPANY_NAME "Aulamix" BUNDLE_ID "com.aulamix.l2limiter" PLUGIN_MANUFACTURER_CODE Aulx PLUGIN_CODE L2Lm FORMATS ${formats} PRODUCT_NAME "L2 Limiter" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 if(MSVC)
  target_compile_options(L2Limiter PRIVATE /O2)
 else()
  target_compile_options(L2Limiter PRIVATE -O2 -fno-strict-aliasing)
 endif()
 target_sources(L2Limiter PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
 target_compile_definitions(L2Limiter PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(L2Limiter PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 add_executable(plugin_tests Tests/PluginTests.cpp)
 target_include_directories(plugin_tests PRIVATE Source ${juce_SOURCE_DIR}/modules ${JUCE_SOURCE_DIR}/modules)
 target_link_libraries(plugin_tests PRIVATE L2Limiter)
 add_test(NAME PluginStateAndAudio COMMAND plugin_tests)
endif()
